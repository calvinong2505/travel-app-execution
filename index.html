// ============================================================ // Trip Execution — Backend API (Code.gs) — SLICE 1 // Trips + Milestones // ============================================================ // Built on the proven CORS POC pattern: GET for reads, // text/plain POST for writes, everything returns JSON. // // SETUP: // 1. This is the SAME Apps Script project where you ran // setupExecution() — so EXECUTION_SHEET_ID is already in // Script Properties. Just REPLACE the setup code with this // file (or add it alongside — setupExecution can stay). // 2. Deploy → New deployment → Web app → // Execute as: Me · Who has access: Anyone // 3. Copy the /exec URL — that's the execution API endpoint. // ============================================================ var TZ = 'Asia/Singapore'; // ---- JSON response helper (same as POC) ---- function jsonOutput(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // ---- Spreadsheet access ---- function execSS_() { var id = PropertiesService.getScriptProperties().getProperty('EXECUTION_SHEET_ID'); if (!id) throw new Error('EXECUTION_SHEET_ID not set — run setupExecution() first'); return SpreadsheetApp.openById(id); } function sheetByName_(name) { var sheet = execSS_().getSheetByName(name); if (!sheet) throw new Error('Missing tab: ' + name); return sheet; } // ---- Generic read: returns array of row objects keyed by header ---- function readRows_(name) { var sheet = sheetByName_(name); if (sheet.getLastRow() <= 1) return []; var data = sheet.getDataRange().getValues(); var headers = data[0].map(function (h) { return h.toString().trim(); }); var out = []; for (var i = 1; i < data.length; i++) { if (!data[i][0] && data[i][0] !== 0) continue; // skip blank rows var obj = {}; headers.forEach(function (h, j) { obj[h] = data[i][j]; }); out.push(obj); } return out; } // ---- Append a row from an object, in header order ---- function appendRow_(name, obj) { var sheet = sheetByName_(name); var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()) .getValues()[0].map(function (h) { return h.toString().trim(); }); var row = headers.map(function (h) { return obj.hasOwnProperty(h) ? obj[h] : ''; }); sheet.appendRow(row); SpreadsheetApp.flush(); } // ---- Update a row found by an id column matching a value ---- function updateRowById_(name, idCol, idVal, changes) { var sheet = sheetByName_(name); var data = sheet.getDataRange().getValues(); var headers = data[0].map(function (h) { return h.toString().trim(); }); var idIdx = headers.indexOf(idCol); if (idIdx === -1) return false; for (var i = 1; i < data.length; i++) { if (data[i][idIdx] && data[i][idIdx].toString() === idVal.toString()) { headers.forEach(function (h, j) { if (changes.hasOwnProperty(h)) { sheet.getRange(i + 1, j + 1).setValue(changes[h]); } }); SpreadsheetApp.flush(); return true; } } return false; } // ---- Delete a row found by id ---- function deleteRowById_(name, idCol, idVal) { var sheet = sheetByName_(name); var data = sheet.getDataRange().getValues(); var headers = data[0].map(function (h) { return h.toString().trim(); }); var idIdx = headers.indexOf(idCol); if (idIdx === -1) return false; for (var i = data.length - 1; i >= 1; i--) { if (data[i][idIdx] && data[i][idIdx].toString() === idVal.toString()) { sheet.deleteRow(i + 1); SpreadsheetApp.flush(); return true; } } return false; } // delete ALL rows in a sheet whose column matches a value (for cascade delete) function deleteAllByCol_(name, col, val) { var sheet = sheetByName_(name); if (!sheet) return 0; var data = sheet.getDataRange().getValues(); var headers = data[0].map(function (h) { return h.toString().trim(); }); var idx = headers.indexOf(col); if (idx === -1) return 0; var removed = 0; for (var i = data.length - 1; i >= 1; i--) { if (data[i][idx] && data[i][idx].toString() === val.toString()) { sheet.deleteRow(i + 1); removed++; } } SpreadsheetApp.flush(); return removed; } // delete a trip and cascade to its milestones, itinerary, bookings function deleteTrip_(data) { var tripId = data.trip_id; if (!tripId) return { ok: false, error: 'No trip_id' }; var okTrip = deleteRowById_('Trips', 'trip_id', tripId); var ms = deleteAllByCol_('Milestones', 'trip_id', tripId); var it = deleteAllByCol_('Itinerary', 'trip_id', tripId); var bk = deleteAllByCol_('Bookings', 'trip_id', tripId); return { ok: okTrip, cascade: { milestones: ms, itinerary: it, bookings: bk } }; } // ---- date helpers (work in plain yyyy-MM-dd strings) ---- function fmtDate_(d) { return Utilities.formatDate(d, TZ, 'yyyy-MM-dd'); } function today_() { return fmtDate_(new Date()); } function addDays_(yyyymmdd, n) { var p = yyyymmdd.split('-'); var d = new Date(parseInt(p[0]), parseInt(p[1]) - 1, parseInt(p[2])); d.setDate(d.getDate() + n); return fmtDate_(d); } // Normalise a time cell to "HH:mm". Sheets coerces "10:00" into a // full Date on the 1899-12-30 epoch, so a Date here means we just // want its time-of-day. Plain strings pass through. function cleanTime_(v) { if (v === '' || v === null || v === undefined) return ''; if (v instanceof Date) return Utilities.formatDate(v, TZ, 'HH:mm'); var s = v.toString(); if (s.indexOf('T') !== -1) { // ISO-ish timestamp string — pull the HH:mm out var t = s.split('T')[1]; return t ? t.slice(0, 5) : s; } return s; } // Normalise any cell value to a clean yyyy-MM-dd string. // Handles: real Date objects (from Sheets), ISO strings, plain // yyyy-MM-dd strings, and blanks. This is the fix for dates // rendering as "Mon Aug 03 2026 00:00:00 GMT". function cleanDate_(v) { if (v === '' || v === null || v === undefined) return ''; if (v instanceof Date) return fmtDate_(v); var s = v.toString(); if (s.indexOf('T') !== -1) return s.split('T')[0]; // ISO string if (/^\d{4}-\d{2}-\d{2}$/.test(s)) return s; // already clean var d = new Date(s); // last resort return isNaN(d.getTime()) ? s : fmtDate_(d); } // ============================================================ // ACTIONS // ============================================================ // ---- Trips ---- function getTrips_() { var trips = readRows_('Trips').map(function (t) { t.start_date = cleanDate_(t.start_date); t.end_date = cleanDate_(t.end_date); return t; }); // newest start_date first trips.sort(function (a, b) { return (b.start_date || '') > (a.start_date || '') ? 1 : -1; }); return trips; } function createTrip_(data) { var tripId = makeTripId_(data.destination, data.start_date); appendRow_('Trips', { trip_id: tripId, destination: data.destination || '', start_date: data.start_date || '', end_date: data.end_date || '', status: data.status || 'planning', home_airport: data.home_airport || '', trip_facts: data.trip_facts || '', notes: data.notes || '' }); // Auto-generate the standard milestones var created = autoMilestones_(tripId, data.start_date); return { trip_id: tripId, milestones_created: created }; } function updateTrip_(data) { var ok = updateRowById_('Trips', 'trip_id', data.trip_id, data.changes || {}); return { updated: ok }; } // trip_id format: slug-yyyymm e.g. lisbon-202507 function makeTripId_(destination, startDate) { var slug = (destination || 'trip').toString().toLowerCase() .replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '').slice(0, 24) || 'trip'; var ym = (startDate || today_()).replace(/-/g, '').slice(0, 6); var base = slug + '-' + ym; // ensure uniqueness if same slug+month already exists var existing = readRows_('Trips').map(function (t) { return t.trip_id; }); if (existing.indexOf(base) === -1) return base; var n = 2; while (existing.indexOf(base + '-' + n) !== -1) n++; return base + '-' + n; } // ---- Milestones ---- function getMilestones_(tripId) { var rows = readRows_('Milestones').map(function (m) { m.due_date = cleanDate_(m.due_date); m.done = (m.done === true || m.done === 'TRUE' || m.done === 'true'); m.auto_created = (m.auto_created === true || m.auto_created === 'TRUE' || m.auto_created === 'true'); return m; }); if (tripId) rows = rows.filter(function (m) { return m.trip_id === tripId; }); rows.sort(function (a, b) { return (a.due_date || '') > (b.due_date || '') ? 1 : -1; }); return rows; } // The standard milestone set, dates worked back from start_date function autoMilestones_(tripId, startDate) { var start = startDate || today_(); var set = [ { name: 'Book flights', due: today_() }, // flag immediately { name: 'Book accommodation', due: addDays_(start, -30) }, { name: 'Reserve activities', due: addDays_(start, -14) }, { name: 'Airport transfer / transport', due: addDays_(start, -7) }, { name: 'Pre-trip checklist (passport, insurance, eSIM, currency, adapter, offline maps, screenshots)', due: addDays_(start, -7) } ]; set.forEach(function (m) { // never suggest a date in the past — clamp to today var due = (m.due < today_()) ? today_() : m.due; appendRow_('Milestones', { milestone_id: Utilities.getUuid(), trip_id: tripId, name: m.name, due_date: due, done: false, auto_created: true, source_stop_id: '' }); }); return set.length; } function addMilestone_(data) { var id = Utilities.getUuid(); appendRow_('Milestones', { milestone_id: id, trip_id: data.trip_id, name: data.name || '', due_date: data.due_date || '', done: false, auto_created: false, source_stop_id: data.source_stop_id || '', description: data.description || '' }); return { milestone_id: id }; } // One-time: add a 'description' column to Milestones if missing. // Safe to run repeatedly. Run once from the editor after pasting. function migrateMilestoneDescription() { var sheet = sheetByName_('Milestones'); var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0] .map(function (h) { return h.toString().trim(); }); if (headers.indexOf('description') === -1) { sheet.getRange(1, sheet.getLastColumn() + 1).setValue('description'); } return { ok: true, headers: headers }; } function updateMilestone_(data) { var ok = updateRowById_('Milestones', 'milestone_id', data.milestone_id, data.changes || {}); return { updated: ok }; } function deleteMilestone_(data) { var ok = deleteRowById_('Milestones', 'milestone_id', data.milestone_id); return { deleted: ok }; } // Cross-trip deadline view: all not-done milestones, soonest first function getDeadlines_() { var trips = {}; getTrips_().forEach(function (t) { trips[t.trip_id] = t.destination; }); var rows = getMilestones_(null).filter(function (m) { return !m.done; }); rows.forEach(function (m) { m.destination = trips[m.trip_id] || m.trip_id; }); rows.sort(function (a, b) { return (a.due_date || '') > (b.due_date || '') ? 1 : -1; }); return rows; } // ============================================================ // ITINERARY (Slice 2) // ============================================================ function getItinerary_(tripId) { var rows = readRows_('Itinerary').map(function (it) { it.date = cleanDate_(it.date); it.time = cleanTime_(it.time); it.est_cost = parseFloat(it.est_cost) || 0; it.route_mins = parseFloat(it.route_mins) || 0; it.route_cost = parseFloat(it.route_cost) || 0; it.day = parseInt(it.day) || 0; // parse the checklist JSON into a real array it.checklist = parseChecklist_(it.checklist_json); delete it.checklist_json; return it; }); if (tripId) rows = rows.filter(function (it) { return it.trip_id === tripId; }); // sort by day, then time rows.sort(function (a, b) { if (a.day !== b.day) return a.day - b.day; return (a.time || '') > (b.time || '') ? 1 : -1; }); return rows; } function parseChecklist_(json) { if (!json) return []; try { var arr = JSON.parse(json); return Array.isArray(arr) ? arr : []; } catch (e) { return []; } } function addItineraryItem_(data) { var id = Utilities.getUuid(); var checklist = data.checklist || []; appendRow_('Itinerary', { item_id: id, trip_id: data.trip_id, day: data.day || 1, date: data.date || '', time: data.time || '', name: data.name || '', type: data.type || 'sightseeing', status: data.status || 'idea', est_cost: data.est_cost || '', currency: data.currency || 'SGD', checklist_json: JSON.stringify(checklist), notes: data.notes || '', route_mode: data.route_mode || '', route_mins: data.route_mins || '', route_cost: data.route_cost || '', booking_id: data.booking_id || '', source_stop_id: data.source_stop_id || '' }); return { item_id: id }; } function updateItineraryItem_(data) { var changes = data.changes || {}; // if the checklist was passed as an array, store it as JSON if (changes.checklist) { changes.checklist_json = JSON.stringify(changes.checklist); delete changes.checklist; } var ok = updateRowById_('Itinerary', 'item_id', data.item_id, changes); return { updated: ok }; } function deleteItineraryItem_(data) { var ok = deleteRowById_('Itinerary', 'item_id', data.item_id); return { deleted: ok }; } // Toggle one checklist item's done state (read JSON, flip, write back) function toggleChecklistItem_(data) { var sheet = sheetByName_('Itinerary'); var rows = sheet.getDataRange().getValues(); var headers = rows[0].map(function (h) { return h.toString().trim(); }); var idIdx = headers.indexOf('item_id'); var clIdx = headers.indexOf('checklist_json'); for (var i = 1; i < rows.length; i++) { if (rows[i][idIdx] && rows[i][idIdx].toString() === data.item_id.toString()) { var list = parseChecklist_(rows[i][clIdx]); if (list[data.index]) { list[data.index].done = !list[data.index].done; sheet.getRange(i + 1, clIdx + 1).setValue(JSON.stringify(list)); SpreadsheetApp.flush(); return { updated: true, checklist: list }; } return { updated: false, error: 'Index out of range' }; } } return { updated: false, error: 'Item not found' }; } // Trip itinerary summary: confirmed-vs-total + planned cost function getItinerarySummary_(tripId) { var items = getItinerary_(tripId); var booked = items.filter(function (it) { return it.status === 'booked'; }).length; var planned = items.reduce(function (s, it) { return s + (it.est_cost || 0); }, 0); return { total: items.length, booked: booked, pct: items.length ? Math.round(booked / items.length * 100) : 0, planned_cost: Math.round(planned * 100) / 100 }; } // ============================================================ // BOOKINGS (Slice 3) // ============================================================ function getBookings_(tripId) { var rows = readRows_('Bookings').map(function (b) { b.date = cleanDate_(b.date); b.time = cleanTime_(b.time); b.end_time = cleanTime_(b.end_time); b.gate_known = (b.gate_known === true || b.gate_known === 'TRUE' || b.gate_known === 'true'); return b; }); if (tripId) rows = rows.filter(function (b) { return b.trip_id === tripId; }); rows.sort(function (a, b) { if ((a.date || '') !== (b.date || '')) return (a.date || '') > (b.date || '') ? 1 : -1; return (a.time || '') > (b.time || '') ? 1 : -1; }); return rows; } function saveBooking_(data) { if (data.booking_id) { var ok = updateRowById_('Bookings', 'booking_id', data.booking_id, bookingFields_(data)); return { updated: ok, booking_id: data.booking_id }; } var id = Utilities.getUuid(); var obj = bookingFields_(data); obj.booking_id = id; obj.trip_id = data.trip_id; appendRow_('Bookings', obj); if (data.item_id) { updateRowById_('Itinerary', 'item_id', data.item_id, { booking_id: id }); } return { created: true, booking_id: id }; } function bookingFields_(data) { return { kind: data.kind || 'other', title: data.title || '', provider: data.provider || '', confirmation_ref: data.confirmation_ref || '', confirmation_link: data.confirmation_link || '', date: data.date || '', time: data.time || '', end_time: data.end_time || '', location: data.location || '', seat: data.seat || '', section: data.section || '', entrance: data.entrance || '', platform_gate: data.platform_gate || '', gate_known: data.gate_known === true || data.gate_known === 'true', notes: data.notes || '' }; } function deleteBooking_(data) { var items = readRows_('Itinerary'); items.forEach(function (it) { if (it.booking_id && it.booking_id.toString() === data.booking_id.toString()) { updateRowById_('Itinerary', 'item_id', it.item_id, { booking_id: '' }); } }); var ok = deleteRowById_('Bookings', 'booking_id', data.booking_id); return { deleted: ok }; } // set just the live gate/platform value (the on-the-day update) function setGate_(data) { var ok = updateRowById_('Bookings', 'booking_id', data.booking_id, { platform_gate: data.platform_gate || '', gate_known: true }); return { updated: ok }; } // One call that returns everything the trip view needs (speed fix) function getTripBundle_(tripId) { return { itinerary: getItinerary_(tripId), milestones: getMilestones_(tripId), bookings: getBookings_(tripId), summary: getItinerarySummary_(tripId) }; } function doGet(e) { var action = (e && e.parameter && e.parameter.action) || 'ping'; try { if (action === 'ping') return jsonOutput({ ok: true, message: 'Execution API alive', serverTime: new Date().toISOString() }); if (action === 'getTrips') return jsonOutput({ ok: true, trips: getTrips_() }); if (action === 'getMilestones') return jsonOutput({ ok: true, milestones: getMilestones_(e.parameter.trip || null) }); if (action === 'getDeadlines') return jsonOutput({ ok: true, deadlines: getDeadlines_() }); if (action === 'getItinerary') return jsonOutput({ ok: true, itinerary: getItinerary_(e.parameter.trip || null) }); if (action === 'getItinerarySummary') return jsonOutput({ ok: true, summary: getItinerarySummary_(e.parameter.trip || null) }); if (action === 'getBookings') return jsonOutput({ ok: true, bookings: getBookings_(e.parameter.trip || null) }); if (action === 'getTripBundle') return jsonOutput({ ok: true, bundle: getTripBundle_(e.parameter.trip || null) }); return jsonOutput({ ok: false, error: 'Unknown GET action: ' + action }); } catch (err) { return jsonOutput({ ok: false, error: err.message }); } } function doPost(e) { var data = {}; try { data = JSON.parse(e.postData.contents); } catch (err) { return jsonOutput({ ok: false, error: 'Bad JSON body' }); } try { if (data.action === 'createTrip') return jsonOutput({ ok: true, result: createTrip_(data) }); if (data.action === 'updateTrip') return jsonOutput({ ok: true, result: updateTrip_(data) }); if (data.action === 'addMilestone') return jsonOutput({ ok: true, result: addMilestone_(data) }); if (data.action === 'updateMilestone') return jsonOutput({ ok: true, result: updateMilestone_(data) }); if (data.action === 'deleteMilestone') return jsonOutput({ ok: true, result: deleteMilestone_(data) }); if (data.action === 'addItineraryItem') return jsonOutput({ ok: true, result: addItineraryItem_(data) }); if (data.action === 'updateItineraryItem') return jsonOutput({ ok: true, result: updateItineraryItem_(data) }); if (data.action === 'deleteItineraryItem') return jsonOutput({ ok: true, result: deleteItineraryItem_(data) }); if (data.action === 'toggleChecklistItem') return jsonOutput({ ok: true, result: toggleChecklistItem_(data) }); if (data.action === 'saveBooking') return jsonOutput({ ok: true, result: saveBooking_(data) }); if (data.action === 'deleteBooking') return jsonOutput({ ok: true, result: deleteBooking_(data) }); if (data.action === 'deleteTrip') return jsonOutput(deleteTrip_(data)); if (data.action === 'setGate') return jsonOutput({ ok: true, result: setGate_(data) }); return jsonOutput({ ok: false, error: 'Unknown POST action: ' + data.action }); } catch (err) { return jsonOutput({ ok: false, error: err.message }); } }
